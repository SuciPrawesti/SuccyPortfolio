InventoryX – Flutter Inventory Management System

A Modern Mobile Inventory Solution

A comprehensive Flutter-based inventory management system with role-based access control, real-time tracking, and intuitive data visualization for efficient business operations.

This page showcases the full UI/UX as a visual walkthrough due to free hosting limitations. All key features are explained below.

Project Highlights

InventoryX is a mobile inventory management system developed with Flutter that provides businesses with a powerful tool to track and manage their inventory efficiently. The system features a clean, minimalist design with responsive layouts for all device sizes.

The application implements a role-based architecture with distinct interfaces for administrators and regular users, ensuring appropriate access levels and functionality segregation. Built with Flutter and PHP backend, InventoryX delivers a cross-platform solution with real-time data synchronization.

Key Features

Role-Based Access

Secure authentication with different interfaces for admin and regular users, with appropriate permission levels for each role.

Inventory Management

Complete tracking of items including stock levels, categories, and item details with image support for visual reference.

Employee Management

Centralized management of employee accounts with role assignment and profile management capabilities.

Stock Transactions

Track all incoming and outgoing inventory items with detailed records including timestamps and responsible personnel.

Reporting

Generate comprehensive reports on inventory levels, transaction history, and stock movement trends.

Responsive Design

Fully responsive interface that adapts to different screen sizes from smartphones to tablets with optimal user experience.

Real-time Sync

Instant data synchronization between mobile app and backend database for up-to-date inventory information.

Advanced Search

Powerful search functionality with filters to quickly locate specific items or transactions in large inventories.

// login.dart
import 'package:flutter/material.dart';
import 'home_page.dart';

class LoginPage extends StatefulWidget {
  const LoginPage({super.key});

  @override
  State<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends State<LoginPage> {
  final _formKey = GlobalKey<FormState>();
  final _usernameController = TextEditingController();
  final _passwordController = TextEditingController();
  bool _isObscure = true;
  bool _isLoading = false;
  String _errorMessage = '';

  @override
  void dispose() {
    _usernameController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _login() async {
    if (!_formKey.currentState!.validate()) return;

    setState(() {
      _isLoading = true;
      _errorMessage = '';
    });

    // Simulate network delay
    await Future.delayed(const Duration(milliseconds: 800));

    try {
      final username = _usernameController.text.trim();
      final password = _passwordController.text.trim();

      if (username == 'admin' && password == 'admin123') {
        Navigator.pushReplacement(
          context,
          MaterialPageRoute(builder: (context) => const HomePage(isAdmin: true)),
        );
      } else if (username == 'user' && password == 'user123') {
        Navigator.pushReplacement(
          context,
          MaterialPageRoute(builder: (context) => const HomePage(isAdmin: false)),
        );
      } else {
        setState(() => _errorMessage = 'Username atau password salah');
      }
    } finally {
      if (mounted) {
        setState(() => _isLoading = false);
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: Colors.white,
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(24),
          child: Form(
            key: _formKey,
            child: Column(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                const Icon(
                  Icons.lock_person,
                  size: 80,
                  color: Color(0xFF6A9C89),
                ),
                const SizedBox(height: 32),
                Text(
                  'Selamat Datang',
                  style: Theme.of(context).textTheme.headlineSmall?.copyWith(
                        fontWeight: FontWeight.bold,
                        color: Colors.black87,
                      ),
                ),
                const SizedBox(height: 8),
                Text(
                  'Silakan masuk ke akun Anda',
                  style: TextStyle(color: Colors.grey[600]),
                ),
                const SizedBox(height: 32),
                TextFormField(
                  controller: _usernameController,
                  decoration: InputDecoration(
                    labelText: 'Username',
                    prefixIcon: const Icon(Icons.person_outline),
                    border: OutlineInputBorder(
                      borderRadius: BorderRadius.circular(8),
                    ),
                    filled: true,
                    fillColor: Colors.grey[50],
                  ),
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'Username harus diisi';
                    }
                    return null;
                  },
                ),
                const SizedBox(height: 16),
                TextFormField(
                  controller: _passwordController,
                  obscureText: _isObscure,
                  decoration: InputDecoration(
                    labelText: 'Password',
                    prefixIcon: const Icon(Icons.lock_outline),
                    suffixIcon: IconButton(
                      icon: Icon(
                        _isObscure ? Icons.visibility_off : Icons.visibility,
                      ),
                      onPressed: () {
                        setState(() => _isObscure = !_isObscure);
                      },
                    ),
                    border: OutlineInputBorder(
                      borderRadius: BorderRadius.circular(8),
                    ),
                    filled: true,
                    fillColor: Colors.grey[50],
                  ),
                  validator: (value) {
                    if (value == null || value.isEmpty) {
                      return 'Password harus diisi';
                    }
                    if (value.length < 6) {
                      return 'Password minimal 6 karakter';
                    }
                    return null;
                  },
                ),
                if (_errorMessage.isNotEmpty)
                  Padding(
                    padding: const EdgeInsets.only(top: 16),
                    child: Text(
                      _errorMessage,
                      style: TextStyle(color: Colors.red[600]),
                    ),
                  ),
                const SizedBox(height: 24),
                SizedBox(
                  width: double.infinity,
                  height: 50,
                  child: ElevatedButton(
                    onPressed: _isLoading ? null : _login,
                    style: ElevatedButton.styleFrom(
                      backgroundColor: const Color(0xFF6A9C89),
                      foregroundColor: Colors.white,
                      shape: RoundedRectangleBorder(
                        borderRadius: BorderRadius.circular(8),
                      ),
                    ),
                    child: _isLoading
                        ? const CircularProgressIndicator(
                            color: Colors.white,
                            strokeWidth: 2,
                          )
                        : const Text(
                            'MASUK',
                            style: TextStyle(fontWeight: FontWeight.bold),
                          ),
                  ),
                ),
                const SizedBox(height: 16),
                TextButton(
                  onPressed: () {
                    // Forgot password action
                  },
                  child: Text(
                    'Lupa password?',
                    style: TextStyle(color: Colors.grey[600]),
                  ),
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}
